python -m influencer_hub.cli digest --json --hours 6 # any window, machine-readable

python -m influencer_hub.cli backup --out /var/backups/hub --keep 14
python -m influencer_hub.cli restore /var/backups/hub/hub-20261011-0900.sqlite3   # asks first; --yes to do it

curl -u admin:$DASHBOARD_ADMIN_PASSWORD /api/metrics   # Prometheus text for the VM's monitor
